\section{What is probability, really?}

\begin{frame}{Probability in antiquity}
    \centering
    \begin{quote}
    ``That is probable which for the most part usually comes to pass, or which is a part of the ordinary beliefs of mankind, or which contains in itself some resemblance to these qualities, whether such resemblance be true or false.''
    \end{quote}
    
    -- Cicero, ca. 80 BCE
\end{frame}

\begin{frame}{The classical formulation}
    The probability of an event $A$ is defined as
    $$
    \prob(A) = \frac{\text{\# of ways $A$ can occur}}{\text{\# of possible outcomes}}
    $$
    \pause\newline

    \textbf{Pros}
    
    Intuitive, works well for dice, card games, etc.
    \pause\newline

    \textbf{Cons}
    
    Doesn't generalize; what if not all outcomes are equally likely?
\end{frame}

\begin{frame}{The frequency formulation}
     The most common definition is in terms of \textbf{relative frequency},
     \pause
     
     $$
     \prob(A) = \underset{n \rightarrow \infty}{\lim} \frac{n_A}{n}
     $$
     
     where $n_A$ is the number of occurrences of $A$.
     \pause

     \textbf{Pros}
     \begin{itemize}[<+->]
         \item Matches intuitions about many physical systems
         \item Empirically measurable in controlled experiments
     \end{itemize}
     \pause

     \textbf{Cons}
     \begin{itemize}[<+->]
         \item Doesn't allow assignment of probabilities to single, rare, or past events
         \item Based on circular reasoning
     \end{itemize}
     
\end{frame}

\begin{frame}{The subjective formulation}
    Probability is a merely an expression of \textbf{belief}
    \pause\newline

    Assigned probabilities correspond to ``betting odds'' on outcomes
    \pause\newline

    Must follow some basic rules in order to be consistent
    \pause\newline

    \textbf{Pros}

    Universally applicable and logically coherent
    \pause

    \textbf{Cons}

    How can such probabilities be verified? How can we know they are \textit{useful}? 
    
\end{frame}

\begin{frame}{Probability as logic}

    Richard Cox (1979) and Edwin Jaynes (1995) showed how probability can be viewed as a \textit{system of logic} for reasoning under uncertainty
    \pause\newline

    Let $A$ and $B$ denote sets of outcomes or, more generally, \textit{propositions}
    \pause\newline

    $A, B \equiv A \cap B$ is the \textit{conjunction} (logical AND)
    \pause\newline

    $A + B \equiv A \cup B$ is the \textit{disjunction} (logical OR)
    \pause\newline

    $\bar{A}$ is the logical \textit{complement} (NOT)
\end{frame}

\begin{frame}{Probability as logic}
    We can derive three basic rules:
    \pause

    \textbf{Sum rule}

    $$ \prob(A\mid B) + \prob(A\mid\bar{B}) = 1$$
    \pause
    \textbf{Product rule}

    $$ \prob(A,B) = \prob(A\mid B) \prob(B) $$
    \pause
    \textbf{Generalized sum rule}

    $$ \prob(A + B) = \prob(A) + \prob(B) - \prob(A,B)$$
\end{frame}

\begin{frame}{Bayes' theorem}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{align*}
                \action<+->{\prob(A,B) &= \prob(B,A)} \\
                \action<+->{\prob(A \mid B)P(B) &= \prob(B\mid A)P(A)} \\
                \action<+->{\prob(A \mid B) &= \frac{\prob(B\mid A)P(A)}{P(B)}}
            \end{align*}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{figure}
                \centering
                \includegraphics[width=0.8\textwidth]{figs/Thomas-Bayes-1024x1061.jpg}
                \caption*{The Reverend Thomas Bayes}
            \end{figure}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Statistical inference via Bayes' rule}
    Let $\mathcal{H}$ be a hypothesis and $\mathcal{D}$ be some observed data
    \pause\newline

    Bayes' theorem tells us that we can make \textbf{inferences} about $\mathcal{H}$ \textit{given} $\mathcal{D}$,

    $$
    \prob(\mathcal{H} \mid \mathcal{D}) = \frac{\prob(\mathcal{D}\mid \mathcal{H})\prob(\mathcal{H})}{\prob(\mathcal{D})}\,.
    $$
    \pause

    $\prob(\mathcal{H})$ is the \textbf{prior} probability
    \pause

    $\prob(\mathcal{D} \mid \mathcal{H})$ is the \textbf{sampling} probability (or \textit{likelihood})
    \pause

    $\prob(\mathcal{H} \mid \mathcal{D})$ is the \textbf{posterior} probability
\end{frame}

\section{A tale of two statisticians}

\begin{frame}{The Frequentist vs. The Bayesian}
    \textbf{The Frequentist} strictly follows the \textit{frequency} definition of probability
    \pause

    Under this interpretation, $\prob(\mathcal{H})$ isn't necessarily well defined
    \pause

    Thus, it only make sense to reason about $\prob(\mathcal{D} \mid \mathcal{H})$
    \pause\newline
    
    \textbf{The Bayesian} follows the more general \textit{subjective} or \textit{logical} definitions
    \pause

    Probability is simply a quantitative measure of \textit{uncertainty}
    \pause

    For the Bayesian, frequency-based probability is a special case
\end{frame}

\begin{frame}{Cliché but helpful: Disease testing}
    Let $\mathcal{D} = \text{positive}$ be the test result for a disease
    \pause\newline

    Suppose the \textbf{sensitivity} of the test is $\prob(\text{positive} \mid \text{sick}) = 99\%$ and the \textbf{false positive rate} is $\prob(\text{positive} \mid \text{not sick}) = 5\%$
    \pause\newline

    How likely is it that the patient has the disease?
    \pause\newline

    Answer: We don't have enough information!
    
\end{frame}

\begin{frame}{Cliché but helpful: Disease testing}
    Suppose the disease is known to affect 1 in 100 people
    \pause\newline

    In a group of \textbf{10,000} people, \textbf{100} would be expected to have the disease
    \pause

    99\% of the \textbf{100 sick} people would be expected to test positive
    \pause

    5\% of the \textbf{9900 healthy} people would also test positive
    \pause\newline

    What is the probability that someone is sick given a positive test?
    \pause
    $$
    \prob(\text{sick} \mid \text{positive}) = \frac{99}{990/2 + 99} \approx 0.17 \times 100 = 17\%
    $$
    
\end{frame}

\begin{frame}{Cliché but helpful: Disease testing}
    \begin{table}[]
        \centering
        \begin{tabular}{l|c|c|c}
             \toprule
             $\mathcal{H}$ & $\prob(\mathcal{H})$ & $\prob(\text{positive} \mid \mathcal{H})$ & $\prob(\text{positive}\mid
             \mathcal{H})\prob(\mathcal{H})$ \\
             \midrule
             sick & 0.01 & 0.99 & 0.0099  \\
             not sick & 0.99 & 0.05 & 0.0495 \\
             \bottomrule
        \end{tabular}
    \end{table}

    \begin{align*}
    \prob(\text{sick} \mid \text{positive}) &= \frac{\prob(\text{positive} \mid \text{sick})\prob(\text{sick})}{\prob(\text{positive})} \\
    &= \frac{0.099}{0.0099 + 0.099} \approx 0.17
    \end{align*}
    \pause

    Failing to account for $\prob(\text{sick})$ is called \textbf{base rate neglect}
\end{frame}

\begin{frame}{The more general case}
    % Our two statisticians would so far be in agreement
    % \pause\newline

    Suppose we have a regression problem $\mathbf{y} = \beta\mathbf{x}$
    \pause\newline

    Our hypothesis is now about the unknown parameter $\beta$
    \pause\newline

    For the Bayesian, nothing really changes:
    $$
    p(\beta \mid \mathbf{y}) = \frac{p(\mathbf{y}\mid \beta)p(\beta)}{\int p(\mathbf{y}\mid\beta^\prime)p(\beta^\prime)d\beta^\prime}
    $$
    \pause

    But the Frequentist would object that $\beta$ cannot have a probability
\end{frame}

\begin{frame}{Null hypothesis significance testing}

\begin{enumerate}
    \item Define your experiment, along with the relevant \textbf{null} ($\mathcal{H}_0$) and (implicit) \textbf{alternative} ($\mathcal{H}_A$) hypotheses
    \newline
    \item Collect your data
    \newline
    \item Compute the relevant \textit{test statistic} $\hat{s}$ from the data
    \newline
    \item Compute the so-called ``$p$-value'' $p = \prob(s > \hat{s} \mid \mathcal{H}_0)$, i.e. the probability of observing a statistic \textit{more extreme} than what you saw
    \newline
    \item Reject $\mathcal{H}_0$ if $p < \alpha$ where $\alpha$ is an arbitrary ``significance'' level.
\end{enumerate}

\end{frame}

\begin{frame}{Criticisms of NHST}

1. The $p$-value, $p = \prob(s > \hat{s} \mid \mathcal{H}_0)$, is often misinterpreted
\pause

\begin{itemize}[<+->]
    \item It is \textbf{not} the probability of the null hypothesis being ``true''
     \newline
    \item It is \textbf{not} the probability that your data was produced by ``random chance''
     \newline
    \item It does \textbf{not} imply anything about the size of the effect (e.g., $\beta$)
     \newline
    \item It says nothing about the alternative hypothesis
\end{itemize}

\end{frame}

\begin{frame}{Problems with NHST}

2. The null hypothesis \textbf{is never true}
\pause\newline

\begin{itemize}[<+->]
    \item When is the ``true'' linear relationship between two values ever exactly zero?
    \newline
    \item When is the ``true'' difference between two groups every exactly zero?
     \newline
    \item Why should we think that such simple models are ``true'' in the first place?
\end{itemize}

\end{frame}

\begin{frame}{Problems with NHST}

2a. ``Significance'' of the p-value depends on the sample size
\pause\newline

For linear regression, the t-statistic is calculated as
\begin{equation*}
t = \frac{\hat{\beta} - \beta_0}{\text{SE}(\hat{\beta})}
\end{equation*}
where the standard error $\text{SE}(\hat{\beta})$ increases with $n$
\pause\newline

This means that, as $n \rightarrow \infty$, \textbf{all effects become significant}

\end{frame}

\begin{frame}{Criticisms of NHST}

3. NHST performs inference about a test statistic $\tilde{s}$ based on \textbf{hypothetical data that was never observed}
\pause\newline

% The null distribution $\prob(\tilde{s} \mid H_0)$ is a \textbf{distribution over the test statistic}

\begin{quote}
\noindent``What the use of p-values implies, therefore, is that a hypothesis that may be true may be rejected because it has not predicted observable results that have not occurred. This seems a remarkable procedure.''

\centering -- Harold Jeffreys (1961, p. 385)    
\end{quote}

\end{frame}

\begin{frame}{Criticisms of NHST}

4. Statistical test procedures make \textbf{non-obvious assumptions}
\pause

\textbf{Student's t-test assumptions}

\begin{itemize}
\item The sample mean is normally distributed
\item The sample variance is $\chi^2$ distributed (or the samples are normally distributed)
\item The sample mean and sample variance are independent
\item The variances of both populations are equal (relaxed for Welch's t-test)
\end{itemize}


\end{frame}

\begin{frame}{Criticisms of NHST}

5. NHST inferences are based on a \textbf{logical fallacy}
\pause\newline

Observing an unlikely event/sample is not evidence against a hypothesis
\pause\newline

We have to compare how well the data supports $\mathcal{H}_A$ vs. $\mathcal{H}_0$
\pause\newline

Unlikely data under the null is not evidence for an alternative

\end{frame}

\begin{frame}{What about confidence intervals?}

Confidence intervals are generally preferable to p-values
\pause

However, they are still easy to misinterpret (Morey et al. 2015):

\begin{enumerate}[<+->]
    \item A particular 95\% confidence interval \textbf{does not} have a 95\% probability of containing the true parameter value
    \newline
    \item Smaller confidence intervals \textbf{do not} imply "more precise" results
    \newline
    \item Confidence intervals \textbf{do not} necessarily correspond to plausible or likely parameter values
\end{enumerate}

\end{frame}

\begin{frame}{Remarks on correlation analysis}
    Linear (Pearson) correlation is one of the \textbf{most widely used} statistical analyses
    \pause

    It is also one of the \textbf{least interesting}
    \pause

    Before analyzing pairwise correlations, consider the following:
    \begin{itemize}[<+->]
        \item Are any of the variables \textbf{categorical}?
        \item Do pairwise scatter plots show clear \textbf{nonlinear} patterns?
        \item Do any of the variables show clear signs of \textbf{non-normality} (e.g., skewed/heavy-tailed distributions)?
        \item Are any of the variables \textbf{mathematically related} \textit{a priori} (e.g., percentages)?
    \end{itemize}
\end{frame}

    Institutional and educational inertia
    \newline\pause

\begin{frame}{Recommendations: The Don'ts}

\textbf{Do not} conflate ``statistical significance'' with scientific significance
\pause\newline

\textbf{Do not} selectively show results based on $p$-values or confidence intervals
\pause\newline

\textbf{Do not} highlight certain results as "statistically significant" based on their $p$-values or confidence intervals

\textbf{Do not} apply ``cookie-cutter'' statistical analyses without thinking carefully about assumptions

\end{frame}

\begin{frame}{Recommendations: The Dos}

\textbf{Do} be as transparent as possible when reporting statistical results
\pause\newline

\textbf{Do} prefer problem-specific error metrics and goodness of fit measures like $R^2$ to $p$-values
\pause\newline

\textbf{Do} use resampling or out-of-sample testing to assess robustness of your results
\pause\newline

\textbf{Do} prefer Neyman-Pearson hypothesis testing or Bayesian analyses over NHST

\end{frame}

\begin{frame}{When to still be frequentist}

Repeated sampling is a feature of the problem
\pause\newline

No meaningful prior information is available
\pause\newline

Sampling distribution and experimental design are well controlled
\pause\newline

Nonparametric methods for large, high-dimensional datasets

\end{frame}

\begin{frame}{When to be Bayesian}

Small, sparse, or incomplete data
\pause\newline

Meaningful prior information is available
\pause\newline

Ill-posed inverse problems
\pause\newline

Models of interest are complex, nonlinear, or hierarchical

\end{frame}

\section{Bayesian inference}

\begin{frame}{Statistical analysis as model building}
    Much of ``classical'' statistical analysis is usually \textbf{descriptive}
    \newline\pause

    In contrast, Bayesian modeling is usually \textbf{generative}
    \newline\pause

    Generative models aim not only to make inferences but to capture key elements of the \textbf{data-generating process}
    \newline\pause

    Recall that probability distributions encode \textbf{uncertainty}
    \begin{itemize}
        \item \textbf{Unobserved variables} $\rightarrow$ Epistemic uncertainty (limited information)
        \item \textbf{Observed variables} $\rightarrow$ Aleatoric uncertainty (unexplained variability)
    \end{itemize}
\end{frame}

\begin{frame}{Building generative models}
    Suppose we want to model soil temperature
    \pause
    \begin{align*}
        T_\text{soil} &\sim \mathcal{N}(\mu=\bar{T}_\text{soil} + \beta T_{\text{air}},\, \Sigma = \varepsilon^2 I)\, \\
        \bar{T}_\text{soil} &\sim \mathcal{N}(\mu=0,\, \sigma^2 = 100)\, \\
        \beta &\sim \mathcal{N}(\mu=0,\, \sigma^2=1)\, \\
        \log\varepsilon &\sim \mathcal{N}|(\mu = 0,\, \sigma^2=1)
    \end{align*}

    The $\beta$ parameter relates $T_\text{soil}$ and $T_\text{air}$
    \pause

    We could make this model \textit{fully generative} by also modeling $T_\text{air}$
    \pause
\end{frame}

\begin{frame}{Bayesian inference}

Bayes' rule gives a way to condition the \textbf{unobserved} variables in the model on the \textbf{observed} variables (data):
$$
p(\boldsymbol{\theta} \mid \mathbf{y}, \mathbf{x}) \propto p(\mathbf{y} \mid \boldsymbol{\theta},\mathbf{x}) p(\boldsymbol{\theta}\mid\mathbf{x})
$$

In the previous example, $\boldsymbol{\theta} = \left[\bar{T}_{\text{soil}}, \beta, \varepsilon \right]$, $\mathbf{y} = T_{\text{soil}}$, and $\mathbf{x} = T_{\text{air}}$
\pause\newline

Once we have a summary of $p(\theta  \mid  \mathbf{y},\mathbf{x})$, we can ask straightforward questions like:
\newline

What is the probability that the linear effect (slope) $\beta > 0$?

\end{frame}

\begin{frame}{Maximum likelihood estimation (MLE)}


Have you ever fitted an OLS line or optimized the mean squared error?
\pause\newline

Then congratulations, you're familiar with MLE!
\pause\newline

MLE aims to maximize the likelihood function $p(\mathbf{y} \mid \boldsymbol{\theta})$ w.r.t $\boldsymbol{\theta}$.
\pause\newline

Typically, this is done via the negative log likelihood:
$$
\mathcal{L}(\boldsymbol{\theta}) = -\log p(\mathbf{y} \mid \boldsymbol{\theta})
$$

and solving $\frac{\partial \mathcal{L}}{\partial \boldsymbol{\theta}} = 0$ for $\boldsymbol{\theta}$.


\end{frame}

\begin{frame}{Maximum likelihood estimation (MLE)}

Common loss functions you may have used:

Mean squared error, $\frac{1}{N}\sum_{i=1}^N (\hat{y}_i - y_i)^2$
\pause\newline

Mean absolute error, $\frac{1}{N}\sum_{i=1}^N \lvert \hat{y}_i - y_i\rvert$
\pause\newline

Cross entropy, $-\frac{1}{N}\sum_{i=1}^N y_i\log \hat{y_i} + (1-y_i)\log(1-\hat{y}_i)$
\pause\newline

All are forms of maximum likelihood

\end{frame}

\begin{frame}{Maximum a posteriori (MAP) estimation}

Maximum a posteriori (MAP) estimation maximizes $p(\boldsymbol{\theta}  \mid  \mathbf{y}) \propto p(\mathbf{y}  \mid  \boldsymbol{\theta})p(\boldsymbol{\theta})$ with respect to $\boldsymbol{\theta}$.
\pause\newline

Or equivalently, minimizes:
$$
\mathcal{L}_{\mathbf{y}}(\boldsymbol{\theta}) = -\log p(\mathbf{y} \mid \boldsymbol{\theta}) - \log p(\boldsymbol{\theta})
$$
\pause\newline

Notice that maximum likelihood estimation (MLE) is just a special case of MAP.


\end{frame}

\begin{frame}{Estimating the posterior distribution}


MAP provides a point estimate of $\boldsymbol{\theta}$
\pause\newline

In Bayesian inference, are (usually) interested in estimating the full distribution $p(\boldsymbol{\theta} \mid \mathbf{y})$
\pause\newline

The goal is evaluate \textbf{expectations} of the form:

$$
\mathbb{E}[f(\boldsymbol{\theta})] = \int_\Theta f(\boldsymbol{\theta})p(\boldsymbol{\theta} \mid \mathbf{y})d\boldsymbol{\theta}
$$

Unfortunately, that's pretty hard!
% $$
% p(\theta  \mid  \mathbf{y}) = \frac{p(\mathbf{y}  \mid  \theta)p(\theta)}{\int_\Theta p(\mathbf{y} \mid \theta)p(\theta)d\theta}
% $$

\end{frame}

\begin{frame}{Markov Chain Monte Carlo}


Enter... Markov Chain Monte Carlo! (MCMC)
\pause\newline

Basic idea: Sample numerically via a random walk in the space of $\boldsymbol{\theta}$
\pause\newline

In the limit of infinite steps, the path of the walk will follow $p(\boldsymbol{\theta} \mid \mathbf{y})$
\pause\newline

Unfortunately, this is often quite inefficient
\pause\newline

Modern state-of-the-art samplers are typically variants \textbf{Hamiltonian Monte Carlo} (HMC) which use gradient information


\end{frame}

\begin{frame}{Key limitations and challenges}
Bayesian inference provides a flexible and intuitive framework for solving inverse problems and building generative models...

\begin{itemize}[<+->]
    \item Choosing good priors can be difficult
    \newline
    \item MCMC is computationally expensive, especially for copmlex models
    \newline
    \item Practical analysis of the inference results can be time consuming
    \newline
    \item Software support
\end{itemize}

\end{frame}
